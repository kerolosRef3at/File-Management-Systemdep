المرحلة 1: كشف الثغرات الحرجة (CRITICAL VULNERABILITIES)
1.1 Hardcoded Credentials (CWE-798) — CRITICAL
الملف: js/shared/mockData.js سطر 9

المطلوب:

افتح js/shared/mockData.js
احذف حقل password: "Admin123" من السطر 9
تأكد أنه لا يوجد أي password field في أي عنصر من mockUsers

javascript

1.3 Password Verification Always Returns True (CWE-287) — CRITICAL
الملف: js/shared/services.js سطر 149

javascript

// الكود المصاب:
return String(password).trim().length >= 3;  // أي باسورد 3 حروف = صحيح!
المشكلة: دالة verifyPassword() لو الـ API فشل بترجع true لأي باسورد طوله 3 حروف أو أكثر. ده معناه إن عمليات الحذف المحمية بالباسورد مش محمية فعلا.

المطلوب: في دالة verifyPassword() (سطر 133-150)، غيّر الـ fallback:

javascript

async verifyPassword(password) {
    if (!password || String(password).trim() === '') return false;
    const currentUser = this.getCurrentUser();
    const username = currentUser?.username || localStorage.getItem('aitu_username') || '';
    if (!username) return false;
    try {
        const controller = new AbortController();
        const timer = setTimeout(() => controller.abort(), 4000);
        const res = await fetchAPI('/api/Auth/verify-password', {
            method: 'POST',
            body: JSON.stringify({ username, password }),
            signal: controller.signal
        });
        clearTimeout(timer);
        if (res && (res.success || res.valid || res.status === 200)) return true;
        return false;
    } catch (e) {
        // If API is unavailable, deny verification for security
        console.error('Password verification failed - API unreachable');
        return false;
    }
},

المطلوب:

أنشئ دالة escapeHTML في js/shared/utils.js:
javascript

export function escapeHTML(str) {
    if (str === null || str === undefined) return '';
    return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#x27;');
}
ابحث في كل ملفات JS عن أي innerHTML أو template literal يحتوي على بيانات ديناميكية مثل file.name, user.username, dept.name, course.title وغيرها
لف كل قيمة ديناميكية بـ escapeHTML()

المرحلة 4: أمان المصادقة والجلسات (Auth and Session Security)

الملف: js/shared/services.js — getCurrentUser()

المشكلة: الكود بيفك الـ JWT ويقرأ البيانات بدون ما يتحقق من صلاحية التوكن (exp claim).

المطلوب: في دالة getCurrentUser() أضف تحقق من exp claim:

javascript

// Check token expiry
if (decoded.exp && decoded.exp < Math.floor(Date.now() / 1000)) {
    console.warn('Token expired, clearing session');
    this.logout();
    return null;
}
